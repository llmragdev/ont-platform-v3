def calculate_accuracy_simple(expected, actual):
    """
    Gemini LLM judge와 동일한 로직

    - "RAG", "인덱싱" 등 기술 내용 → 0점
    - 모호한 경우 → 25~75점
    """
    expected_str = str(expected).lower()
    actual_str = str(actual).lower()
    
    if "관련이 없습니다" in expected_str:
        if "관련이 없습니다" in actual_str:
            return 100
        elif any(word in actual_str for word in ["해당 카테고리", "문서"]):
            return 75
        elif any(word in actual_str for word in ["rag", "인덱싱"]):
            return 0

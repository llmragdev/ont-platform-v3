===> systemd 폴더 없음. 

아래 링크 참조하여 바로 /etc/systemd/system에서 mariadb.service 작성 
https://bakch92-github-io.vercel.app/blog/db/1.%20MariaDB%20%EB%B0%94%EC%9D%B4%EB%84%88%EB%A6%AC%20%EC%84%A4%EC%B9%98%20%EB%B0%8F%20%EC%84%A4%EC%A0%95

cd /etc/systemd/system
$vi mariadb.service

[Unit]
Description=MariaDB Database Server
After=network.target

[Service]
Type=forking
User=mariadb
Group=mariadb
ExecStart=/usr/local/mariadb/bin/mysqld_safe --datadir='/var/lib/mysql/data'
ExecStop=/usr/local/mariadb/bin/mysqladmin shutdown

[Install]
WantedBy=multi-user.target

10. 마리아DB 서비스 시작
systemctl start mariadb.service

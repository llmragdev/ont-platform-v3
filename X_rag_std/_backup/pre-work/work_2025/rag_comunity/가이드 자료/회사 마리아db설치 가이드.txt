* DB설치<MariaDB> *
[root@localhost ~]# cd /usr/local/src
[root@localhost src]# tar xvfz mariadb-10.3.18-linux-x86_64.tar.gz
[root@localhost src]# mv mariadb-10.3.18-linux-x86_64 /usr/local/
[root@localhost src]# cd /usr/local
[root@localhost local]# mv mariadb-10.3.18-linux-x86_64/ mysql

# CentOS7의 경우 mysql 계정이 생성되어 있어야 한다고 합니다.
[root@localhost local]# adduser mysql
[root@localhost local]# cat /etc/passwd | grep mysql

# mysql 디렉토리 소유자 및 그룹사용자 변경
(/usr/local 경로에서 명령어 입력)
[root@localhost local]# chown mysql.mysql -R mysql
[root@localhost local]# cp /usr/local/src/my.cnf /etc/my.cnf

# mysql.server 파일 복사/덮어쓰기 및 각 실행 관련 파일 소유자 및 그룹사용자 변경, 실행 파일 권한 설정
[root@localhost local]# cp /usr/local/mysql/support-files/mysql.server /etc/init.d/mysqld
[root@localhost local]# chown mysql.mysql /etc/init.d/mysqld
[root@localhost local]# chown mysql.mysql /etc/my.cnf
[root@localhost local]# chmod 750 /etc/init.d/mysqld


[root@localhost local]# vi /etc/my.cnf  - (확인)

[mysqld]
datadir=/usr/local/mysql/data

밑에 아래의 부분 추가
init_connect=SET collation_connection = utf8_general_ci
init_connect=SET NAMES utf8
character-set-server = utf8
collation-server = utf8_general_ci

# mysql 실행 파일 수정 및 저장 (basedir 및 datadir 수정)
[root@localhost local]# vi /etc/init.d/mysqld
(아래와 같이 수정)
basedir=/usr/local/mysql
datadir=/usr/local/mysql/data


# mysql datadir 소유자 및 그룹사용자 변경
[root@localhost local]# chown mysql.mysql -R /usr/local/mysql/data

# libaio 패키지 설치
[root@localhost local]# yum install libaio*
[root@localhost local]# yum install libtinfo*

# mysql (mariadb) install 작업
[root@localhost local]# cd /usr/local/mysql/scripts
[root@localhost local]# ./mysql_install_db --basedir=/usr/local/mysql/ --datadir=/usr/local/mysql/data --defaults-file=/etc/my.cnf

import jamo

def tokenize_korean(text: str) -> Set[str]:
    """한글 토큰화 강화"""
    tokens = set()
    
    # 1. 전체 단어
    tokens.add(text.lower())
    
    # 2. 음절 단위
    for char in text:
        if '가' <= char <= '힣':
            tokens.add(char)
    
    # 3. 초성 분해 (선택적)
    for char in text:
        if '가' <= char <= '힣':
            cho = jamo.decompose(char)[0]
            tokens.add(cho)
    
    # 4. 부분 문자열 (3자 이상)
    for i in range(len(text) - 2):
        tokens.add(text[i:i+3])
    
    return tokens

**Task 2-4: Gemini LLM Judge 자동 채점**

**파일:** `scripts/auto_score_24qa.py` (신규)

**작업:**
```python
import anthropic
import json

# Gemini API (또는 Anthropic API) 사용
client = anthropic.Anthropic()

async def auto_score_answers(eval_results):
    """24개 답변 자동 채점"""
    
    scoring_prompt = """
주어진 질문, 예상 답변, 실제 답변을 평가하세요.

평가 기준:
1. 핵심 항목 포함 (40점): 예상 답변의 주요 내용이 포함되었는가?
2. 과정 설명 (30점): 구체적이고 명확한가?
3. 용어 정확성 (20점): 기술 용어가 정확한가?
4. 문장 명확성 (10점): 이해하기 쉬운가?

응답 형식:
{{
  "score": 85,
  "reasoning": "...",
  "strengths": ["...", "..."],
  "weaknesses": ["...", "..."]
}}
"""
    
    for result in eval_results:
        prompt = f"""
질문: {result['question']}

예상 답변:
{result['expected']}

실제 답변:
{result['actual']}

{scoring_prompt}
"""
        
        response = client.messages.create(
            model="claude-opus-4-8",
            max_tokens=500,
            messages=[{"role": "user", "content": prompt}]
        )
        
        score_data = json.loads(response.content[0].text)
        result['score'] = score_data['score']
        result['scoring'] = score_data
    
    return eval_results

# 실행
eval_results = json.load(open('eval_results_24qa.json'))
scored_results = await auto_score_answers(eval_results)

# 결과 저장
with open('eval_results_24qa_scored.json', 'w') as f:
    json.dump(scored_results, f, indent=2)
```

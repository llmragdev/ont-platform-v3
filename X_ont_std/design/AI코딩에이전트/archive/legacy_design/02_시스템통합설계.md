      <label className="block text-sm font-medium mb-2">

from fastapi import APIRouter, Depends, HTTPException
from app.services.ai_coding_agent import AICodingAgentService
from app.models.schemas import QueryRequest, QueryResponse
from app.dependencies import get_current_user

router = APIRouter(prefix="/api/ai-coding-agent", tags=["AI Coding Agent"])

@router.post("/query")
async def execute_query(
    request: QueryRequest,
    user = Depends(get_current_user)
) -> QueryResponse:
    """
    자연어 쿼리를 받아서 코드 생성, 검증, 실행까지 수행
    """
    service = AICodingAgentService()
    
    try:
        # 1. Intent Detection
        intent = await service.detect_intent(request.query)
        
        # 2. Schema Analysis
        schema = await service.analyze_schema(intent)
        
        # 3. Code Generation
        code = await service.generate_code(intent, schema, request.language)
        
        # 4. Validation
        validation = await service.validate_code(code, request.language, schema, intent)
        
        if not validation.overall_valid:
            return QueryResponse(
                status="VALIDATION_FAILED",
                validation=validation,
                code=code
            )
        
        # 5. Execution
        execution = await service.execute_code(
            code, 
            request.language, 
            user
        )
        
        # 6. Explanation
        explanation = await service.explain_results(
            execution, 
            intent, 
            request.query
        )
        
        # 7. Audit Logging
        await service.log_query(
            user_id=user.id,
            query=request.query,
            intent=intent,
            code=code,
            execution=execution
        )
        
        return QueryResponse(
            status="SUCCESS",
            intent=intent,
            code=code,
            validation=validation,
            execution=execution,
            explanation=explanation
        )
    
    except Exception as e:
        return QueryResponse(
            status="ERROR",
            error=str(e)
        )

### 3️⃣ **온톨로지 서비스 통합**

```python
# app/services/ai_coding_agent.py
from app.services.ontology_service import OntologyService
from app.services.workflow_service import WorkflowService
from app.services.data_service import DataService

class AICodingAgentService:
    def __init__(self):
        self.ontology = OntologyService()
        self.workflow = WorkflowService()
        self.data = DataService()
        self.intent_detector = IntentDetector(ontology=self.ontology)
        self.schema_analyzer = SchemaAnalyzer(ontology=self.ontology)
        self.sql_generator = SQLCodeGenerator()
        self.sparql_generator = SPARQLCodeGenerator()
        self.python_generator = PythonCodeGenerator()
        self.validator = CodeValidator()
        self.executor = CodeExecutor(
            sql_engine=self.data.sql_engine,
            python_sandbox=self.data.python_sandbox,
            sparql_endpoint=self.ontology.sparql_endpoint
        )
        self.explainer = ResultExplainer()
    
    async def detect_intent(self, query: str) -> IntentSchema:
        """자연어를 IntentSchema로 변환"""
        return self.intent_detector.detect(query)
    
    async def analyze_schema(self, intent: IntentSchema) -> SchemaMap:
        """온톨로지 스키마 분석"""
        return self.schema_analyzer.analyze(intent)
    
    async def generate_code(self, 
                           intent: IntentSchema, 
                           schema: SchemaMap,
                           language: str) -> str:
        """코드 생성"""
        if language == "sql":
            return self.sql_generator.generate(intent, schema)
        elif language == "sparql":
            return self.sparql_generator.generate(intent, schema)
        elif language == "python":
            return self.python_generator.generate(intent, schema)
        else:
            raise ValueError(f"Unknown language: {language}")
    
    async def validate_code(self,
                           code: str,
                           language: str,
                           schema: SchemaMap,
                           intent: IntentSchema) -> ValidationResult:
        """코드 검증"""
        return self.validator.validate(code, language, schema, intent)
    
    async def execute_code(self,
                          code: str,
                          language: str,
                          user) -> ExecutionResult:
        """코드 실행 (권한 검증 포함)"""
        return self.executor.execute(code, language, user)
    
    async def explain_results(self,
                             execution: ExecutionResult,
                             intent: IntentSchema,
                             original_query: str) -> ExplanationResult:
        """결과 분석 및 설명"""
        return self.explainer.explain(execution, intent, original_query)
    
    async def log_query(self,
                       user_id: str,
                       query: str,
                       intent: IntentSchema,
                       code: str,
                       execution: ExecutionResult):
        """감사 로그 기록"""
        await self.data.audit_log.create({
            "user_id": user_id,
            "activity": "ai-coding-query",
            "query": query,
            "intent": intent.dict(),
            "code": code,
            "status": execution.status,
            "execution_time": execution.execution_time,
            "rows_returned": execution.rows_returned
        })
    
    async def save_as_workflow(self, query_id: str, workflow_name: str, user_id: str):
        """쿼리를 워크플로우로 저장"""
        # 쿼리 이력에서 코드 조회
        query_history = await self.data.query_history.get(query_id)
        
        # 워크플로우 생성
        workflow = {
            "name": workflow_name,
            "description": query_history.query,
            "skill_id": f"ai-code-{query_history.language}",
            "inputs": {
                "code": query_history.code
            },
            "scheduled": False
        }
        
        return await self.workflow.create_workflow(workflow, user_id)
    
    async def schedule_execution(self, query_id: str, schedule: str, user_id: str):
        """쿼리를 정기적으로 실행하도록 스케줄 설정"""
        query_history = await self.data.query_history.get(query_id)
        
        # WorkflowService의 schedule 기능 활용
        return await self.workflow.schedule_workflow(
            query_history.workflow_id,
            schedule,
            user_id
        )
```

-- ai_coding_queries: 모든 쿼리 이력 저장
CREATE TABLE ai_coding_queries (
    id UUID PRIMARY KEY,
    user_id UUID NOT NULL REFERENCES users(id),
    
    -- 입력
    query TEXT NOT NULL,
    language VARCHAR(20),  -- sql, sparql, python, workflow
    
    -- 생성된 결과
    intent JSONB,
    code TEXT,
    
    -- 검증 결과
    validation JSONB,
    
    -- 실행 결과
    execution JSONB,
    
    -- 설명
    explanation JSONB,
    
    -- 메타데이터
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    executed_at TIMESTAMP,
    
    INDEX idx_user_id (user_id),
    INDEX idx_created_at (created_at)
);

-- ai_code_cache: 자주 실행되는 쿼리 캐싱
CREATE TABLE ai_code_cache (
    id UUID PRIMARY KEY,
    code_hash VARCHAR(64) UNIQUE NOT NULL,
    code TEXT NOT NULL,
    language VARCHAR(20),
    schema_hash VARCHAR(64),
    
    result JSONB,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    accessed_at TIMESTAMP,
    access_count INT DEFAULT 1,
    
    ttl_seconds INT DEFAULT 86400,
    
    INDEX idx_code_hash (code_hash),
    INDEX idx_accessed_at (accessed_at)
);

-- ai_code_workflows: 저장된 워크플로우로부터 생성된 쿼리
CREATE TABLE ai_code_workflows (
    workflow_id UUID PRIMARY KEY REFERENCES workflows(id),
    query_id UUID REFERENCES ai_coding_queries(id),
    
    scheduled_cron VARCHAR(100),
    last_execution_at TIMESTAMP,
    next_execution_at TIMESTAMP,
    
    execution_count INT DEFAULT 0,
    success_count INT DEFAULT 0,
    error_count INT DEFAULT 0
);

            description="출력 언어 (sql, python, sparql)",

        )

            type="string",

            type="object",
            description="검증 결과"
        )

    async def execute(self, inputs: Dict) -> Dict:
        from app.services.ai_coding_agent import AICodingAgentService
        
        service = AICodingAgentService()
        
        intent = await service.detect_intent(inputs["query"])
        schema = await service.analyze_schema(intent)
        code = await service.generate_code(
            intent,
            schema,

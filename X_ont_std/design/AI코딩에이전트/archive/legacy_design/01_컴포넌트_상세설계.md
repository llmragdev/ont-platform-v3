### 1️⃣ **Intent Detector** (의도 파악)

#### 목적
자연어 입력을 분석하여 사용자가 수행하려는 작업의 의도를 파악

#### 입력
```
사용자 입력: "지난 3개월 매출이 100만 이상인 고객들의 이름과 매출을 보여줘"
온톨로지 컨텍스트: Customer, Sales 엔티티
```

#### 처리 로직
```python
class IntentDetector:
    def __init__(self, ontology, llm):
        self.ontology = ontology
        self.llm = llm
        
    def detect(self, query: str) -> IntentSchema:
        # 1. 엔티티 추출
        entities = self.llm.extract_entities(
            query, 
            available_entities=self.ontology.entities
        )
        
        # 2. 필터 조건 추출
        filters = self.llm.extract_filters(
            query,
            entity_properties=self.ontology.get_entity_properties(entities)
        )
        
        # 3. 의도 분류
        intent_type = self._classify_intent(query)  # SELECT, CREATE, UPDATE, ANALYZE, AUTOMATE
        
        # 4. 출력 필드 추출
        output_fields = self.llm.extract_output_fields(query, entities)
        
        # 5. 신뢰도 계산
        confidence = self._calculate_confidence(entities, filters, intent_type)
        
        return IntentSchema(
            type=intent_type,
            entities=entities,
            filters=filters,
            output_fields=output_fields,
            confidence=confidence
        )
    
    def _classify_intent(self, query: str) -> str:
        keywords = {
            "SELECT": ["보여줘", "조회", "어디있어", "누구"],
            "ANALYZE": ["분석", "비교", "추세", "패턴"],
            "CREATE": ["만들어", "생성", "추가"],
            "UPDATE": ["수정", "변경", "업데이트"],
            "AUTOMATE": ["매일", "자동으로", "스케줄", "설정해"]
        }
        
        for intent, keywords_list in keywords.items():
            if any(kw in query for kw in keywords_list):
                return intent
        return "SELECT"
```

#### 출력
```json
{
  "type": "SELECT",
  "entities": ["Customer", "Sales"],
  "filters": [
    {
      "entity": "Sales",
      "property": "sale_date",
      "operator": ">=",
      "value": "2026-03-14"  // 3개월 전
    },
    {
      "entity": "Sales",
      "property": "amount",
      "operator": ">=",
      "value": 1000000
    }
  ],
  "output_fields": [
    {"entity": "Customer", "property": "customer_name"},
    {"entity": "Sales", "property": "amount"}
  ],
  "confidence": 0.95
}
```

- 정확도: 90%+

### 2️⃣ **Schema Analyzer** (스키마 이해)

#### 목적
온톨로지에서 필요한 엔티티, 관계, 속성 정보를 추출하여 정규화

#### 입력
```
IntentSchema {
  entities: ["Customer", "Sales"],
  filters: [...]
}
온톨로지 데이터베이스
```

#### 처리 로직
```python
class SchemaAnalyzer:
    def __init__(self, ontology_service):
        self.ontology = ontology_service
        
    def analyze(self, intent: IntentSchema) -> SchemaMap:
        schema_map = SchemaMap()
        
        # 1. 엔티티 정보 수집
        for entity_name in intent.entities:
            entity = self.ontology.get_entity(entity_name)
            schema_map.entities[entity_name] = {
                "properties": entity.properties,
                "relationships": entity.relationships,
                "indexes": entity.indexes,
                "primary_key": entity.primary_key
            }
        
        # 2. 관계 분석
        for entity in intent.entities:
            relationships = self.ontology.get_relationships(entity)
            for rel in relationships:
                schema_map.relationships[f"{entity}-{rel.target}"] = {
                    "type": rel.type,  # hasMany, hasOne, etc
                    "cardinality": rel.cardinality,
                    "join_key": rel.join_key
                }
        
        # 3. 인덱스 추출
        for entity in intent.entities:
            indexes = self.ontology.get_indexes(entity)
            schema_map.indexes[entity] = indexes
        
        # 4. 계산 필드 확인
        computed = self.ontology.get_computed_fields(intent.entities)
        schema_map.computed_fields = computed
        
        return schema_map
```

```json

class SQLCodeGenerator:
    def __init__(self, dialect="postgresql"):
        self.dialect = dialect
        self.templates = load_sql_templates()
    
    def generate(self, intent: IntentSchema, schema: SchemaMap) -> str:
        # 1. SELECT 절 생성
        select_clause = self._build_select(intent.output_fields, schema)
        
        # 2. FROM 절 생성
        from_clause = self._build_from(intent.entities, schema)
        
        # 3. JOIN 절 생성
        join_clauses = self._build_joins(intent.entities, schema)
        
        # 4. WHERE 절 생성
        where_clause = self._build_where(intent.filters, schema)
        
        # 5. GROUP BY / ORDER BY 생성
        group_by = self._build_group_by(intent)
        order_by = self._build_order_by(intent)
        
        # 완성
        query = f"""
        {select_clause}
        {from_clause}
        {' '.join(join_clauses)}
        {where_clause}
        {group_by}
        {order_by}
        """
        
        return query.strip()

# SPARQL 생성도 유사
class SPARQLCodeGenerator:
    def generate(self, intent: IntentSchema, schema: SchemaMap) -> str:
        # PREFIX 정의
        prefixes = self._build_prefixes(schema)
        
        # SELECT 절
        select_vars = self._extract_variables(intent.output_fields)
        
        # WHERE 패턴 생성
        where_patterns = self._build_where_patterns(
            intent.entities,
            intent.filters,
            schema
        )
        
        query = f"""
        {prefixes}
        SELECT {' '.join(select_vars)}
        WHERE {{
          {where_patterns}
        }}
        """
        
        return query.strip()

class CodeValidator:
    def __init__(self, schema_analyzer, performance_analyzer):
        self.schema = schema_analyzer
        self.perf = performance_analyzer
    
    def validate(self, code: str, code_type: str, 
                 schema: SchemaMap, intent: IntentSchema) -> ValidationResult:
        result = ValidationResult()
        
        # 1. 문법 검사
        result.syntax_valid = self._check_syntax(code, code_type)
        
        # 2. 스키마 검증
        tables = self._extract_tables(code)
        columns = self._extract_columns(code)
        result.schema_valid = self._validate_schema(tables, columns, schema)
        
        # 3. 성능 분석
        result.performance_score = self.perf.analyze(code, schema)
        result.estimated_time = self.perf.estimate_execution_time(code)
        
        # 4. 보안 검사
        result.security_valid = self._check_security(code)
        if not result.security_valid:
            result.security_issues = self._find_sql_injection_risks(code)
        
        # 5. 논리 검증
        result.logic_valid = self._validate_logic(code, intent, schema)
        
        # 6. 종합 판정
        result.overall_valid = (
            result.syntax_valid and
            result.schema_valid and
            result.security_valid and
            result.logic_valid
        )
        
        return result

class ValidationResult:
    def __init__(self):
        self.syntax_valid: bool
        self.schema_valid: bool
        self.performance_score: float  # 1.0 ~ 5.0
        self.estimated_time: float  # ms
        self.security_valid: bool
        self.security_issues: List[str]
        self.logic_valid: bool
        self.warnings: List[str]
        self.overall_valid: bool
```

### 5️⃣ **Executor** (코드 실행)

#### 목적
검증된 코드를 안전한 샌드박스에서 실행

#### 입력
```
검증된 코드
ValidationResult (VALID만 실행)
사용자 권한
```

#### 처리 로직
```python
class CodeExecutor:
    def __init__(self, sql_engine, python_sandbox, sparql_endpoint):
        self.sql = sql_engine
        self.python = python_sandbox
        self.sparql = sparql_endpoint
    
    def execute(self, code: str, code_type: str, 
                user: User, validation: ValidationResult) -> ExecutionResult:
        
        # 1. 권한 검증
        if not validation.overall_valid:
            raise ValueError("Code validation failed")
        
        if not self._check_permissions(user, code):
            raise PermissionError(f"User {user.id} cannot access this data")
        
        # 2. 타임아웃 설정
        timeout = min(validation.estimated_time * 2, 30000)  # 최대 30초
        
        # 3. 실행
        start_time = time.time()
        try:
            if code_type == "sql":
                result = self.sql.execute(code, timeout=timeout)
            elif code_type == "python":
                result = self.python.execute(code, timeout=timeout)
            elif code_type == "sparql":
                result = self.sparql.execute(code, timeout=timeout)
            else:
                raise ValueError(f"Unknown code type: {code_type}")
            
            execution_time = time.time() - start_time
            
            # 4. 결과 캐싱 (같은 쿼리는 24시간 재사용)
            cache_key = hash(code + user.id)
            self._cache.set(cache_key, result, ttl=86400)
            
            return ExecutionResult(
                status="SUCCESS",
                data=result,
                execution_time=execution_time,
                rows_returned=len(result) if isinstance(result, list) else 1,
                cached=False
            )
        
        except TimeoutError:
            return ExecutionResult(
                status="TIMEOUT",
                error="Execution exceeded 30 seconds",
                execution_time=timeout
            )
        
        except Exception as e:
            return ExecutionResult(
                status="ERROR",
                error=str(e),
                execution_time=execution_time
            )

class ExecutionResult:
    def __init__(self, status: str, **kwargs):
        self.status = status  # SUCCESS, ERROR, TIMEOUT
        self.data = kwargs.get('data')
        self.error = kwargs.get('error')
        self.execution_time = kwargs.get('execution_time', 0)
        self.rows_returned = kwargs.get('rows_returned', 0)
        self.cached = kwargs.get('cached', False)
```

### 6️⃣ **Explainer** (결과 설명)

#### 목적
실행 결과를 분석하여 사용자 친화적 설명과 인사이트 생성

#### 입력
```
ExecutionResult
Intent
원본 사용자 질문
```

#### 처리 로직
```python
class ResultExplainer:
    def __init__(self, llm, analytics_engine):
        self.llm = llm
        self.analytics = analytics_engine
    
    def explain(self, execution: ExecutionResult, 
                intent: IntentSchema, query: str) -> ExplanationResult:
        
        if execution.status != "SUCCESS":
            return ExplanationResult(
                summary=f"쿼리 실행 실패: {execution.error}",
                insights=[],
                suggestions=["코드를 수정하고 다시 실행해보세요"]
            )
        
        data = execution.data
        
        # 1. 기본 요약
        summary = self._generate_summary(data, intent)
        
        # 2. 인사이트 분석
        insights = []
        
        # 2-1. 통계 분석
        if len(data) > 0:
            stats = self.analytics.compute_statistics(data)
            insights.append(f"총 {len(data)}개의 레코드가 조회됨")
            
            # 숫자형 컬럼에 대한 통계
            for col in self._get_numeric_columns(data):
                insights.append(
                    f"{col}: 평균 {stats[col]['mean']:.0f}, "
                    f"최소 {stats[col]['min']:.0f}, "
                    f"최대 {stats[col]['max']:.0f}"
                )
        
        # 2-2. 추세 분석
        if self._has_time_series(data):
            trends = self.analytics.detect_trends(data)
            insights.extend(trends)
        
        # 2-3. 이상치 탐지
        anomalies = self.analytics.detect_anomalies(data)
        if anomalies:
            insights.append(f"⚠️ 이상치 감지: {anomalies}")
        
        # 3. 제안 생성
        suggestions = self._generate_suggestions(data, insights, intent)
        
        return ExplanationResult(
            summary=summary,
            row_count=len(data),
            insights=insights,
            suggestions=suggestions
        )
    
    def _generate_summary(self, data, intent) -> str:
        if len(data) == 0:
            return "조건을 만족하는 데이터가 없습니다"
        
        summary = f"{len(data)}개의 결과를 찾았습니다."
        
        # 주요 값 강조
        if len(data) <= 3:
            top_items = data
        else:
            # 크기 기준으로 정렬 (숫자형 컬럼이 있다면)
            numeric_cols = self._get_numeric_columns(data)
            if numeric_cols:
                sorted_data = sorted(
                    data, 
                    key=lambda x: x[numeric_cols[0]], 
                    reverse=True
                )
                top_items = sorted_data[:3]
            else:
                top_items = data[:3]
        
        summary += "\n상위 3개: "
        for item in top_items:
            summary += f"\n - {item}"
        
        return summary
    
    def _generate_suggestions(self, data, insights, intent) -> List[str]:
        suggestions = []
        
        # 기본 제안
        if "평균" in insights[0] if insights else False:
            suggestions.append("이 고객들의 구매 패턴을 분석해볼까요?")
        
        suggestions.append("상위 고객들의 세부 정보를 보시겠어요?")
        suggestions.append("다음 달 예측 분석을 수행할까요?")
        
        return suggestions

class ExplanationResult:
    def __init__(self, summary: str, insights: List[str], suggestions: List[str], **kwargs):
        self.summary = summary
        self.row_count = kwargs.get('row_count', 0)
        self.insights = insights
        self.suggestions = suggestions
```

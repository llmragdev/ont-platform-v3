```python
# tests/integration/test_query_execution.py
class TestQueryExecution:
    async def test_full_pipeline(self):
        # 1. 의도 분석
        intent = await detector.detect("지난 3개월 매출 > 100만인 고객 목록")
        
        # 2. 스키마 분석
        schema = await analyzer.analyze(intent)
        
        # 3. 코드 생성
        code = await generator.generate(intent, schema, "sql")
        
        # 4. 검증
        validation = await validator.validate(code, "sql", schema, intent)
        assert validation.overall_valid
        
        # 5. 실행
        result = await executor.execute(code, "sql", user)
        assert result.status == "SUCCESS"
        assert len(result.data) > 0
        
        # 6. 설명
        explanation = await explainer.explain(result, intent, original_query)
        assert explanation.summary
```

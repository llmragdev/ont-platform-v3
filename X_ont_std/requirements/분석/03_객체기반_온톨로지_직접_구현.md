# 객체기반 온톨로지 직접 구현

## 1. 목표

이 문서는 특정 상용 플랫폼 없이 객체기반 온톨로지의 핵심 구조를 직접 설계하고 구현하는 방법을 설명합니다. 구현 대상은 작지만, 온톨로지의 핵심 요소인 객체 타입, 객체 인스턴스, 속성, 관계, 액션을 모두 포함합니다.

예제는 Python으로 작성합니다.

## 2. 구현할 개념

온톨로지 엔진의 최소 구성은 다음과 같습니다.

- ObjectType: 객체의 종류
- ObjectInstance: 실제 객체
- PropertyDefinition: 속성 정의
- RelationshipDefinition: 관계 정의
- RelationshipInstance: 객체 간 실제 연결
- ActionDefinition: 객체에 수행할 수 있는 행동
- OntologyRegistry: 전체 모델을 관리하는 레지스트리

## 3. 기본 모델 코드

```python
from dataclasses import dataclass, field
from typing import Any, Callable


@dataclass
class PropertyDefinition:
    name: str
    value_type: type
    required: bool = False


@dataclass
class ObjectType:
    name: str
    properties: dict[str, PropertyDefinition] = field(default_factory=dict)

    def add_property(self, name: str, value_type: type, required: bool = False) -> None:
        self.properties[name] = PropertyDefinition(name, value_type, required)


@dataclass
class ObjectInstance:
    object_id: str
    object_type: ObjectType
    values: dict[str, Any]

    def validate(self) -> None:
        for property_name, definition in self.object_type.properties.items():
            if definition.required and property_name not in self.values:
                raise ValueError(f"Missing required property: {property_name}")

            if property_name in self.values:
                value = self.values[property_name]
                if not isinstance(value, definition.value_type):
                    raise TypeError(
                        f"{property_name} must be {definition.value_type.__name__}"
                    )
```

## 4. 관계 모델 코드

```python
@dataclass
class RelationshipDefinition:
    name: str
    source_type: ObjectType
    target_type: ObjectType


@dataclass
class RelationshipInstance:
    relationship_type: RelationshipDefinition
    source_id: str
    target_id: str
    values: dict[str, Any] = field(default_factory=dict)
```

관계는 단순한 foreign key보다 명시적입니다. 예를 들어 `Customer -> Order` 관계를 `PLACED_ORDER`라고 이름 붙이면, 데이터 탐색과 권한 관리, 액션 설계가 더 쉬워집니다.

## 5. 액션 모델 코드

```python
@dataclass
class ActionDefinition:
    name: str
    object_type: ObjectType
    handler: Callable[[ObjectInstance, dict[str, Any]], Any]

    def execute(self, target: ObjectInstance, payload: dict[str, Any]) -> Any:
        if target.object_type.name != self.object_type.name:
            raise TypeError(
                f"Action {self.name} cannot run on {target.object_type.name}"
            )

        return self.handler(target, payload)
```

액션은 온톨로지를 단순 조회 모델이 아니라 실행 가능한 업무 모델로 만들어 줍니다.

## 6. 레지스트리 구현

```python
class OntologyRegistry:
    def __init__(self) -> None:
        self.object_types: dict[str, ObjectType] = {}
        self.objects: dict[str, ObjectInstance] = {}
        self.relationship_types: dict[str, RelationshipDefinition] = {}
        self.relationships: list[RelationshipInstance] = []
        self.actions: dict[str, ActionDefinition] = {}

    def register_object_type(self, object_type: ObjectType) -> None:
        self.object_types[object_type.name] = object_type

    def create_object(
        self,
        object_id: str,
        object_type_name: str,
        values: dict[str, Any],
    ) -> ObjectInstance:
        object_type = self.object_types[object_type_name]
        instance = ObjectInstance(object_id, object_type, values)
        instance.validate()
        self.objects[object_id] = instance
        return instance

    def register_relationship_type(
        self,
        relationship_type: RelationshipDefinition,
    ) -> None:
        self.relationship_types[relationship_type.name] = relationship_type

    def link(
        self,
        relationship_name: str,
        source_id: str,
        target_id: str,
        values: dict[str, Any] | None = None,
    ) -> RelationshipInstance:
        relationship_type = self.relationship_types[relationship_name]
        source = self.objects[source_id]
        target = self.objects[target_id]

        if source.object_type.name != relationship_type.source_type.name:
            raise TypeError("Invalid source object type")

        if target.object_type.name != relationship_type.target_type.name:
            raise TypeError("Invalid target object type")

        relationship = RelationshipInstance(
            relationship_type=relationship_type,
            source_id=source_id,
            target_id=target_id,
            values=values or {},
        )
        self.relationships.append(relationship)
        return relationship

    def register_action(self, action: ActionDefinition) -> None:
        self.actions[action.name] = action

    def execute_action(
        self,
        action_name: str,
        object_id: str,
        payload: dict[str, Any],
    ) -> Any:
        action = self.actions[action_name]
        target = self.objects[object_id]
        return action.execute(target, payload)

    def find_related(
        self,
        source_id: str,
        relationship_name: str,
    ) -> list[ObjectInstance]:
        result = []

        for relationship in self.relationships:
            if (
                relationship.source_id == source_id
                and relationship.relationship_type.name == relationship_name
            ):
                result.append(self.objects[relationship.target_id])

        return result
```

## 7. 예제 온톨로지 만들기

```python
registry = OntologyRegistry()

customer_type = ObjectType("Customer")
customer_type.add_property("name", str, required=True)
customer_type.add_property("segment", str)
customer_type.add_property("region", str)

order_type = ObjectType("Order")
order_type.add_property("status", str, required=True)
order_type.add_property("amount", float, required=True)

registry.register_object_type(customer_type)
registry.register_object_type(order_type)

placed_order = RelationshipDefinition(
    name="PLACED_ORDER",
    source_type=customer_type,
    target_type=order_type,
)
registry.register_relationship_type(placed_order)
```

## 8. 객체와 관계 생성

```python
customer = registry.create_object(
    object_id="C001",
    object_type_name="Customer",
    values={
        "name": "Alpha Manufacturing",
        "segment": "Enterprise",
        "region": "Seoul",
    },
)

order = registry.create_object(
    object_id="O001",
    object_type_name="Order",
    values={
        "status": "Submitted",
        "amount": 3200.0,
    },
)

registry.link("PLACED_ORDER", "C001", "O001")
```

## 9. 액션 구현

```python
def approve_order(order: ObjectInstance, payload: dict[str, Any]) -> dict[str, Any]:
    if order.values["status"] != "Submitted":
        raise ValueError("Only submitted orders can be approved")

    order.values["status"] = "Approved"
    order.values["approved_by"] = payload["user"]

    return {
        "object_id": order.object_id,
        "status": order.values["status"],
        "approved_by": order.values["approved_by"],
    }


approve_order_action = ActionDefinition(
    name="ApproveOrder",
    object_type=order_type,
    handler=approve_order,
)

registry.register_action(approve_order_action)

result = registry.execute_action(
    action_name="ApproveOrder",
    object_id="O001",
    payload={"user": "analyst@example.com"},
)

print(result)
```

## 10. 관계 탐색

orders = registry.find_related("C001", "PLACED_ORDER")

for item in orders:

## 11. 확장 아이디어

직접 구현한 온톨로지는 다음 방향으로 확장할 수 있습니다.

- 객체와 관계를 데이터베이스에 저장
- 속성별 접근 권한 추가
- 액션 실행 이력 저장
- 관계 탐색을 그래프 데이터베이스로 확장
- 계산 속성 지원
- 이벤트 기반 액션 트리거 추가
- JSON Schema 또는 Pydantic 기반 검증 추가
- REST API 또는 Streamlit UI 연결

## 12. 설계 원칙

객체기반 온톨로지를 직접 만들 때 중요한 점은 데이터 구조보다 업무 의미를 먼저 정의하는 것입니다.

좋은 순서:

1. 업무 객체를 정한다
2. 객체의 핵심 속성을 정한다
3. 객체 간 관계를 명명한다
4. 사용자가 실행해야 하는 액션을 정의한다
5. 원천 데이터와 객체 모델을 매핑한다
6. 권한, 검증, 이력 관리를 추가한다

## 13. 요약

객체기반 온톨로지는 데이터를 업무 객체로 재구성하고, 객체 간 관계와 실행 가능한 액션을 함께 다루는 구조입니다. 직접 구현할 때는 작은 레지스트리부터 시작해 객체 타입, 객체 인스턴스, 관계, 액션을 명확하게 분리하는 것이 좋습니다.
